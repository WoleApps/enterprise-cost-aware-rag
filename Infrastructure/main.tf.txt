resource "aws_opensearchserverless_security_policy" "encryption" {
  name        = "${local.name_prefix}-enc"
  type        = "encryption"
  description = "Encryption policy for RAG vector collection"
  policy = jsonencode({
    Rules = [
      {
        ResourceType = "collection"
        Resource     = ["collection/${local.name_prefix}-vectors"]
      }
    ]
    AWSOwnedKey = true
  })
}

resource "aws_opensearchserverless_security_policy" "network" {
  name        = "${local.name_prefix}-net"
  type        = "network"
  description = "Network policy for RAG vector collection"
  policy = jsonencode([
    {
      Rules = [
        {
          ResourceType = "collection"
          Resource     = ["collection/${local.name_prefix}-vectors"]
        },
        {
          ResourceType = "dashboard"
          Resource     = ["collection/${local.name_prefix}-vectors"]
        }
      ]
      AllowFromPublic = true
    }
  ])
}

resource "aws_opensearchserverless_collection" "vector_store" {
  name        = "${local.name_prefix}-vectors"
  type        = "VECTORSEARCH"
  description = "Vector storage for enterprise RAG embeddings"

  depends_on = [
    aws_opensearchserverless_security_policy.encryption,
    aws_opensearchserverless_security_policy.network
  ]
}

  name        = "${local.name_prefix}-data"

resource "aws_cloudwatch_metric_alarm" "bedrock_token_spike" {
  alarm_name          = "${local.name_prefix}-high-token-consumption"
  comparison_operator = "GreaterThanThreshold"
  evaluation_periods  = 1
  metric_name         = "InputTokenCount"
  namespace           = "AWS/Bedrock"
  period              = 300
  statistic           = "Sum"
  threshold           = var.token_spend_threshold
  alarm_description   = "Triggers if aggregate input tokens cross enterprise cost thresholds within a 5-minute window."
  alarm_actions       = [aws_sns_topic.governance_alerts.arn]

  dimensions = {
    ModelId = "anthropic.claude-3-5-sonnet-20241022-v2:0"
  }
}

resource "aws_cloudwatch_metric_alarm" "bedrock_error_anomaly" {
  alarm_name          = "${local.name_prefix}-error-rate-anomaly"
  comparison_operator = "GreaterThanUpperThreshold"
  evaluation_periods  = 2
  threshold_metric_id = "e1"
  alarm_description   = "Detects sudden anomalies in Bedrock invocation server errors using machine learning bands."
  alarm_actions       = [aws_sns_topic.governance_alerts.arn]

  metric_query {
    id          = "e1"
    expression  = "ANOMALY_DETECTION_BAND(m1, 2)"
    label       = "ErrorRate Anomaly Band"
    return_data = "true"
  }

  metric_query {
    id = "m1"
    metric {
      metric_name = "InvocationServerErrors"
      namespace   = "AWS/Bedrock"
      period      = 300
      stat        = "Sum"
    }
  }
